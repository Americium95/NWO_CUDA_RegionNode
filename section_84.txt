<h3>8.4 0x04/0x10 프레이밍 디코더 — 동기, 상한 복사, ByteBuf 비노출 (공유 포트 디스패처)</h3>
<pre>// AI_Work/PathProtocol/PathPlanFrameDecoder.cs   (제안, 미생성)
public sealed class PathPlanFrameDecoder : ByteToMessageDecoder
{
    // ★ Decode 는 채널 이벤트 루프에서 동기 호출된다. await/block 금지.
    protected override void Decode(IChannelHandlerContext context, IByteBuffer input, List&lt;object&gt; output)
    {
        // (1) opcode 바이트조차 없으면 아무것도 소비하지 않고 반환 → 다음 read 에서 이어진다.
        if (input.ReadableBytes() &lt; 1) return;

        // (2) ★ peek. readerIndex 를 움직이지 않는다 (Dispatcher 분기와 동일한 원칙).
        byte op = input.GetByte(input.ReaderIndex);
        if (op != PathPlanContract.OpcodePathPlan)
        {
            // 0x06 전용 파이프라인에 다른 opcode 가 왔다 = 계약 위반(framing 불가).
            throw new CorruptedFrameException("unexpected opcode on path-plan pipeline: 0x" + op.ToString("X2"));
        }

        // (3) 헤더 12바이트 대기. DotNetty 의 Get* 기본은 <b>BE</b> 이므로 헤더는 위 Codec(LE)을 쓴다.
        if (input.ReadableBytes() &lt; PathPlanContract.HeaderLength) return;
        if (!PathPlanHeaderCodec.TryRead(input.GetReaderSlice(), out var h, out var err))
        {
            if (err == PathPlanDecodeError.Truncated) return;   // 대기
            throw new CorruptedFrameException("bad path-plan header: " + err);
        }

        // (4) 상한 검증. 이 검사가 메모리를 제한하는 유일한 지점이며, 할당보다 먼저 온다.
        if (h.TotalPacketLength &lt; PathPlanContract.MinPacketLength || h.TotalPacketLength &gt; PathPlanContract.MaxPacketLength)
            throw new CorruptedFrameException("totalPacketLength out of range: " + h.TotalPacketLength);

        // (5) v1 은 요청만 받는다. 응답(0x02) 을 서버 포트에서 받는 것은 프로토콜 위반.
        if (h.MessageType != PathPlanContract.MessageTypeRequest)
            throw new CorruptedFrameException("expected request messageType");

        // (6) 아직 안 모였으면 소비하지 않고 반환. 누적 버퍼 상한은 MaxPacketLength 로 제한된다.
        if (input.ReadableBytes() &lt; (int)h.TotalPacketLength) return;

        // (7) 상한을 통과했으므로 NOW 복사한다. 이 byte[] 가 비동기 경계를 넘길 유일한 값이다.
        byte[] raw = new byte[h.TotalPacketLength];
        input.ReadBytes(raw);                                   // 정확히 totalPacketLength 만큼만 소비

        // (8) 본문 길이는 totalPacketLength - 12 로 유일하게 결정된다. 별도 bodyLength 검증이 필요 없다.
        if (raw.Length - PathPlanContract.HeaderLength &lt; PathPlanContract.MinRequestBodyBytes)
            throw new CorruptedFrameException("request body shorter than contract minimum");

        // (9) 불변 객체로 감싸 파이프라인 아래로 내보낸다. IByteBuffer 를 캡처하지 않는다.
        output.Add(PathPlanFrame.Parse(raw));                   // throws → ExceptionCaught → 채널 닫힘(ByteToMessageDecoder 가 버퍼 해제)
    }
    // 파싱 실패 시에도 누적 버퍼가 해제되므로 예외 전파가 새 leak 이 되지 않는다.
    // 길이/버전 위반은 "조용히 버리지 않는다": 채널을 닫아 반쪽짜리 프레임 상태로 남지 않게 한다.
    // ★ TCP 는 느린 공격자가 totalPacketLength 바이트를 보내놓고 멈출 수 있으므로,
    //   채널별 누적 대기 deadline(예 5s)과 최대 누적 바이트 상한을 두는 것을 권장한다.
}

// AI_Work/PathProtocol/PathPlanFrame.cs   (제안, 미생성)
// ★ 완전히 불변 + ByteBuf 비보유. 이렇게 해야 ChannelRead 를 지난 뒤에도 안전하다.
public sealed class PathPlanFrame
{
    public byte[] Bytes { get; }                              // 헤더+본문 전체 복사본
    public ReadOnlySpan&lt;byte&gt; Body =&gt; Bytes.AsSpan(PathPlanContract.HeaderLength);

    public static PathPlanFrame Parse(byte[] raw)
    {
        if (!PathPlanHeaderCodec.TryRead(raw, out var h, out var err)) throw new CorruptedFrameException(err.ToString());
        if (h.TotalPacketLength != (uint)raw.Length) throw new CorruptedFrameException("declared length != actual");
        return new PathPlanFrame(raw);
    }
    // Parse 는 입력 byte[] 를 수정하지 않는다 → 호출자가 가진 프레임은 그대로 불변이다.</pre>
