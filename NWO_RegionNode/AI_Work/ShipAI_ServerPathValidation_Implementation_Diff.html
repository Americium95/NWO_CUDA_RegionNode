3. 전송 계층 계약 (헤더 — 정확히 12바이트)

아래 필드는 본문이 아니라 패킷 헤더다. 길이 프리픽스도 bodyLength 필드도 없다. 헤더가 totalPacketLength를 스스로 갖고 있어 패킷이 자기 경계(self-delimiting)가 된다.

오프셋  크기  타입    필드                 값/설명
  0      1    byte    opcode                0x04 (기존 0x04 계열 확장)
  1      1    byte    subtype               요청 0x10, 응답 0x11
  2      2    uint16  protocolVersion      1  (little-endian)
  4      4    uint32  totalPacketLength    헤더 포함 전체 패킷 바이트 수 (little-endian)
  8      4    uint32  requestId            요청-응답 상관관계 (little-endian)

헤더 크기 = 1 + 1 + 2 + 4 + 4 = 12 bytes   ← 필드 합이 곧 헤더 크기다 (모순 없음)
본문 = packet[12 .. totalPacketLength)
본문 길이 = totalPacketLength - 12        ← 별도 필드가 없으므로 불일치가 원천 불가능

요청 첫 12바이트:  04 10 01 00 | L0 L1 L2 L3 | R0 R1 R2 R3
응답 첫 12바이트:  04 11 01 00 | L0 L1 L2 L3 | R0 R1 R2 R3
                   └op └sub └ver=1 └ LE 총 길이 ┘  └ LE requestId ┘

검증(서버가 수신하는 방향):  opcode == 0x04 and subtype == 0x10
        protocolVersion == 1
        12 <= totalPacketLength <= 1,048,576
        누적 수신 바이트 >= totalPacketLength  → 정확히 totalPacketLength 만큼 소비
경로 프로토콜 판별(공유 19100 스트림):  opcode == 0x04 and subtype ∈ {0x10, 0x11} → 경로 프로토콜
        0x02, 그리고 0x04 의 그 밖 서브타입  → 기존 고정 길이 규칙(8.1 표)

3.2 PathPlanHeaderCodec 의사코드

// AI_Work/PathProtocol/PathPlanHeader.cs   (제안, 미생성)
public readonly struct PathPlanHeader
{
    public byte   Opcode          { get; }   // 0x04
    public byte   Subtype         { get; }   // 0x10/0x11
    public ushort ProtocolVersion { get; }   // 1
    public uint   TotalPacketLength { get; } // 헤더 포함
    public uint   RequestId       { get; }
    public int    BodyLength      => checked((int)TotalPacketLength - PathPlanContract.HeaderLength);
}

public static class PathPlanHeaderCodec
{
    // ★ 엔디안 명시: BitConverter/BinaryPrimitives 는 LE, DotNetty 의 GetUInt32(index) 기본은 BE다.
    //   이 프로토콜은 LE 이므로 GetUInt32LE / SetUInt32LE 계열만 쓴다.
    public static bool TryRead(ReadOnlySpan<byte> src, out PathPlanHeader h, out PathPlanDecodeError err)
    {
        if (src.Length < PathPlanContract.HeaderLength) { h = default; err = PathPlanDecodeError.Truncated; return false; }

        byte   opcode  = src[0];
        byte   subtype = src[1];
        ushort version = BinaryPrimitives.ReadUInt16LittleEndian(src.Slice(2, 2));
        uint   total   = BinaryPrimitives.ReadUInt32LittleEndian(src.Slice(4, 4));
        uint   reqId   = BinaryPrimitives.ReadUInt32LittleEndian(src.Slice(8, 4));

        // 순서 중요: 길이 검사를 opcode 검사보다 먼저 하면 "쓰레기 바이트"를 즉시 잘라낸다.
        if (total < PathPlanContract.MinPacketLength || total > PathPlanContract.MaxPacketLength)
        { h = default; err = PathPlanDecodeError.PacketLengthOutOfRange; return false; }          // 연결 종료 대상
        if (opcode != PathPlanContract.OpcodePathPlan)
        { h = default; err = PathPlanDecodeError.NotPathPlanOpcode; return false; }                 // 위임 대상
        if (subtype != PathPlanContract.SubtypeRequest)
        { h = default; err = PathPlanDecodeError.UnexpectedSubtype; return false; }
        if (version != PathPlanContract.ProtocolVersionV1)
        { h = default; err = PathPlanDecodeError.UnsupportedProtocolVersion; return false; }

        h = new PathPlanHeader(opcode, subtype, version, total, reqId);
        err = PathPlanDecodeError.None;
        return true;
    }

    public static void Write(Span<byte> dst, PathPlanHeader h)
    {
        dst[0] = h.Opcode;
        dst[1] = h.Subtype;
        BinaryPrimitives.WriteUInt16LittleEndian(dst.Slice(2, 2), h.ProtocolVersion);
        BinaryPrimitives.WriteUInt32LittleEndian(dst.Slice(4, 4), h.TotalPacketLength);
        BinaryPrimitives.WriteUInt32LittleEndian(dst.Slice(8, 4), h.RequestId);
    }
}

public enum PathPlanDecodeError : byte
{
    None = 0,
    Truncated,                  // 누적 < 12 → 대기 (오류 아님)
    PacketLengthOutOfRange,     // total < 12 또는 > 1 MiB → 연결 종료
    NotPathPlanOpcode,          // 다른 프로토콜로 위임
    UnexpectedSubtype,          // 서버가 받는 방향은 SubtypeRequest(0x10) 뿐
    UnsupportedProtocolVersion, // version != 1
    BodyLengthMismatch,         // 본문 바이트 수 != total - 12
    FieldTruncated,             // 본문 내 어떤 필드의 남은 바이트가 부족
    CountOutOfRange,            // visitCount / pathPolygonCount 상한 위반
    IntegerOverflow,            // 개수 × 고정 폭 계산이 범위를 넘어섬
    InvalidUtf8,                // 유효하지 않은 UTF-8 또는 NUL 바이트 포함
    TrailingBytes,              // 모든 필드 읽은 뒤 남는 바이트 > 0
    UnknownCode                 // 알 수 없는 decision/reason 값
}

4.1.2 SpanReader — 엄격 경계 커서 (제안, 미생성)

// AI_Work/PathProtocol/SpanReader.cs   (제안, 미생성)
// ★ 규칙: 모든 Try* 는 false 면 커서를 전혀 전진시키지 않는다. 부분 소비 상태를 남기지 않는다.
public ref struct SpanReader
{
    private readonly ReadOnlySpan<byte> _src;
    private int _pos;
    public SpanReader(ReadOnlySpan<byte> src) { _src = src; _pos = 0; }
    public int Position      => _pos;
    public int Remaining     => _src.Length - _pos;
    public bool AtEnd        => Remaining == 0;

    private bool TryTake(int n, out ReadOnlySpan<byte> span)   // n < 0 방어
    {
        if (n < 0 || n > Remaining) { span = default; return false; }
        span = _src.Slice(_pos, n); _pos += n; return true;
    }

    public bool TryReadUInt16(out ushort v) { if (!TryTake(2, out var s)) { v = 0; return false; } v = BinaryPrimitives.ReadUInt16LittleEndian(s); return true; }
    public bool TryReadUInt32(out uint   v) { if (!TryTake(4, out var s)) { v = 0; return false; } v = BinaryPrimitives.ReadUInt32LittleEndian(s); return true; }
    public bool TryReadUInt64(out ulong v) { if (!TryTake(8, out var s)) { v = 0; return false; } v = BinaryPrimitives.ReadUInt64LittleEndian(s); return true; }
    public bool TryReadInt32 (out int    v) { if (!TryTake(4, out var s)) { v = 0; return false; } v = BinaryPrimitives.ReadInt32LittleEndian(s);  return true; }

    // 길이 접두 문자열. 상한을 바이트 수로 먼저 확인하고, 그 다음 UTF-8 유효성과 NUL 부재를 검사한다.
    public bool TryReadUtf8(int maxBytes, out string value, out PathPlanDecodeError err)
    {
        value = string.Empty; err = PathPlanDecodeError.None;
        if (!TryReadUInt16(out ushort len))                { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (len > maxBytes)                                { err = PathPlanDecodeError.CountOutOfRange; return false; }
        if (len > Remaining)                               { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (!TryTake(len, out var raw))                    { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (raw.IndexOf((byte)0) >= 0)                   { err = PathPlanDecodeError.InvalidUtf8; return false; }
        if (!Utf8Strict.IsValid(raw))                      { err = PathPlanDecodeError.InvalidUtf8; return false; }  // 엄격 검사
        value = Utf8Strict.GetString(raw);                 // 이미 검증된 바이트만 문자열화
        return true;
    }
}

// Utf8Strict 는 Encoding.UTF8 를 쓰되 throwOnInvalidBytes: false 로 만들지 않는다.
// (false 로 만들면 깨진 바이트가 U+FFFD 로 치환되어 "유효하지 않은 입력이 조용히 통과"한다.)
// 점진적 디코더(Decoder)나 Rune 순회로 Rune.DecodeFromUtf8 의 Try* 결과를 확인하는 편이 명확하다.

4.1.3 PathPlanRequestDecoder 의사코드

// AI_Work/PathProtocol/PathPlanRequestDecoder.cs   (제안, 미생성)
public static class PathPlanRequestDecoder
{
    // ★ 호출 전제: body.Length == header.TotalPacketLength - 12 이 이미 보장되어 있다(프레이밍 단계).
    //   여기서는 그래도 다시 확인한다(이중 방어). 본 함수는 예외를 던지지 않는다.
    public static bool TryDecode(ReadOnlySpan<byte> body, out PathPlanRequestV1 req, out PathPlanDecodeError err)
    {
        req = default!; err = PathPlanDecodeError.None;
        var r = new SpanReader(body);

        // ── 고정 헤더부 ─────────────────────────────────────────────
        if (!r.TryReadUInt64(out ulong seq))              { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (!r.TryReadUtf8(PathPlanContract.MaxNavMeshVersionBytes, out string version, out err)) return false;
        if (version.Length == 0)                          { err = PathPlanDecodeError.CountOutOfRange; return false; }  // 요청 버전은 비면 무의미
        if (!r.TryReadInt32(out int startPoly))           { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (!r.TryReadInt32(out int destPoly))            { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (!r.TryReadUInt16(out ushort visitCount))      { err = PathPlanDecodeError.FieldTruncated; return false; }

        // ★ 개수 상한을 할당 전에 검사한다. 65535 × 72 는 4.7 MB 라 1 MiB 상한을 넘어도
        //   "잘릴 때까지 읽는다" 는 접근이 여기서 차단된다.
        if (visitCount > PathPlanContract.MaxVisits)    { err = PathPlanDecodeError.CountOutOfRange; return false; }

        // ── visit × K ──────────────────────────────────────────────
        var visits = new List<PathPlanVisitV1>(visitCount);   // visitCount <= 256 이므로 미리 상한 확인됨
        for (int k = 0; k < visitCount; k++)
        {
            if (!r.TryReadUtf8(PathPlanContract.MaxVisitIdBytes, out string vid, out err)) return false;
            if (!r.TryReadInt32(out int  vpoly))          { err = PathPlanDecodeError.FieldTruncated; return false; }
            if (!r.TryReadUInt16(out ushort order))       { err = PathPlanDecodeError.FieldTruncated; return false; }
            // order == k 는 구조 규칙이므로 여기서 확정해도 되지만, 판정 코드 일관성을 위해
            // 검증기로 넘겨 PLAN_INVALID 로 거절시킨다(디코더는 형식만 본다).
            visits.Add(new PathPlanVisitV1(vid, vpoly, order));
        }

        // ── path 통로 ──────────────────────────────────────────────
        if (!r.TryReadUInt32(out uint pathCount))         { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (pathCount < PathPlanContract.MinPathPolygons || pathCount > PathPlanContract.MaxPathPolygons)
        { err = PathPlanDecodeError.CountOutOfRange; return false; }

        // ★ 오버플로 방어: unchecked 한정 안에서 4 * pathCount 를 먼저 계산하면 uint 를 넘어선다.
        //   경로 배치를 "할당 후 채움" 으로 계산해 두 방법 중 하나를 택하고, 다른 하나는 어설프게 쓴다.
        long need = 4L * pathCount;
        if (need > r.Remaining)                         { err = PathPlanDecodeError.FieldTruncated; return false; }  // 잘림

        var path = new int[pathCount];                    // <= 4096, 상한 확인 후 할당
        for (uint i = 0; i < pathCount; i++)
            path[i] = BinaryPrimitives.ReadInt32LittleEndian(body.Slice(r.Position + (int)(4L * i), 4));
        r.Skip((int)need);                                // 스킵은 need <= Remaining 이 확인된 뒤에만

        // ── 꼬리 ───────────────────────────────────────────────────
        if (!r.TryReadUInt64(out ulong computedAt))       { err = PathPlanDecodeError.FieldTruncated; return false; }
        if (!r.AtEnd)                                     { err = PathPlanDecodeError.TrailingBytes; return false; }   // 남는 바이트 = 잘림/오염

        req = new PathPlanRequestV1(seq, version, startPoly, destPoly, visits, path, computedAt);
        return true;
    }
}
이 디코더의 안전 속성 4가지. (1) 모든 개수는 할당 전에 상한 검사된다. (2) 4 * pathCount 같은 산술은 64비트로 수행하고 Remaining과 비교해 버퍼 안임을 증명한 뒤에만 전진한다. (3) Remaining 밖은 절대 읽지 않는다(SpanReader의 TryTake가 보장). (4) 필드를 다 읽은 뒤 남는 바이트가 있으면 거절한다 — 잘린 패킷이 "조용히 성공"할 수 없다.

4.3 PathPlanResponseEncoder 의사코드

// AI_Work/PathProtocol/PathPlanResponseEncoder.cs   (제안, 미생성)
public static class PathPlanResponseEncoder
{
    public static byte[] Encode(in PathPlanValidationV1 v, uint requestId)
    {
        // 1) 문자열을 바이트로. 상한은 인코딩 후 길이로 다시 확인한다(서로 다른 UTF-8 정규화/대용 문자 회피).
        byte[] verBytes = Utf8Strict.GetBytes(v.NavMeshVersion);
        byte[] planBytes= v.ServerPlanId is null ? Array.Empty<byte>() : Utf8Strict.GetBytes(v.ServerPlanId);

        if (verBytes.Length  > PathPlanContract.MaxNavMeshVersionBytes) throw new InvalidOperationException("navMeshVersion too long");
        if (planBytes.Length > PathPlanContract.MaxServerPlanIdBytes)   throw new InvalidOperationException("serverPlanId too long");
        if (verBytes.AsSpan().IndexOf((byte)0) >= 0 || planBytes.AsSpan().IndexOf((byte)0) >= 0)
            throw new InvalidOperationException("NUL in wire string");     // 계약 위반을 조용히 보내지 않는다

        // 2) 길이를 먼저 계산하고, 그 값으로 totalPacketLength 를 쓴다. (뒤에서 고치면 어긋난다)
        int bodyLen = 8 + 1 + 2
                    + 2 + verBytes.Length
                    + 4 + 4
                    + 2 + planBytes.Length;
        int packetLen = PathPlanContract.HeaderLength + bodyLen;

        if (packetLen > PathPlanContract.MaxResponsePacketBytes)      // 방어: 계약 상한(227) 확인
            throw new InvalidOperationException("response packet too large");
        if (packetLen > PathPlanContract.MaxPacketLength)             // 방어: 전송 상한(1 MiB) 확인
            throw new InvalidOperationException("response exceeds 1MiB");

        // 3) 정확히 packetLen 만큼 할당하고 정확히 채운다. 남는 바이트가 없어야 한다.
        var buf = new byte[packetLen];
        var h = new PathPlanHeader(PathPlanContract.OpcodePathPlan, PathPlanContract.SubtypeResponse,
                                   PathPlanContract.ProtocolVersionV1, (uint)packetLen, requestId);
        PathPlanHeaderCodec.Write(buf, h);

        var r = new SpanWriter(buf.AsSpan(PathPlanContract.HeaderLength, bodyLen));
        r.WriteUInt64(v.PlanSequence);
        r.WriteByte((byte)v.Decision);
        r.WriteUInt16((ushort)v.ReasonCode);
        r.WriteUInt16((ushort)verBytes.Length);  verBytes.CopyTo(r.RemainingSpan(verBytes.Length));
        r.WriteUInt32(v.SampledSegments);
        r.WriteUInt32(v.TotalSegments);
        r.WriteUInt16((ushort)planBytes.Length); planBytes.CopyTo(r.RemainingSpan(planBytes.Length));

        if (r.Position != bodyLen) throw new InvalidOperationException("encoder/decoder length drift");  // 자기 검증
        return buf;
    }
}
// SpanWriter 는 SpanReader 의 대칭이며, 모든 Write 는 남은 공간 충분성을 먼저 검사한다.
// Write 는 경계를 넘어도 조용히 잘라 쓰지 않는다 — 예외로 fail-fast 한다(인코더는 서버 내부 버그이므로).
// (성공 응답에서는 예외가 AbsolutePath 로 새지 않도록 핸들러가 fail-closed 응답으로 대체한다.)

코드 7 의 단일 의미 (Plan 4.5 를 그대로 따른다). PATH_DISCONTINUITY(7) 는 오직 "통로상 인접한 두 다각형 id 사이에 공유 포털이 하나도 없다"는 뜻이다. 서버 navmesh 조회(GetPortals)가 빈 목록을 돌려주는 유일한 경우가 이 코드를 만든다. 이 문서의 어느 표·주석·의사코드도 코드 7 에 다른 의미를 붙이지 않는다.
선분 길이 상한/하한은 v1 계약에 존재하지 않는다. 통로 스텝 길이(stepLen/total)는 서버가 계산한 값이며 표본 밴드 가중(7.4)에만 쓰인다 — 거절 사유가 아니다. 어떤 길이 임계값도 reasonCode 를 만들지 않으며, 클라이언트가 보낸 수치도 길이 근거로 쓰지 않는다. 길이 파생 기하 이상(스텝 길이 0 = 두 다각형 중심점이 동일)은 길이 상한/하한 위반이 아니라 통로 구조 위반이므로 PLAN_INVALID(6) 으로 거절한다. 나중에 길이 임계가 필요해지면 기존 번호를 재정의하지 말고 끝에 새 번호를 추가한다.

안정성 규약. reasonCode는 wire 계약이며 재사용·재정의 금지. 신규 사유는 끝에 새 번호를 추가하고 기존 번호는 남겨둔다. 서버는 위 표에 있는 값만 송신한다. 클라이언트는 미할당 reasonCode 또는 decision ∉ {0,1,2,3} 을 받으면 fail-closed로 그 응답을 폐기한다 — 알 수 없는 사유를 "일단 통과"로 해석하지 않는다.

픽스처 블록의 16진수 값은 손으로 계산한 예시이므로 권위 있는 원천이 아니다. 실제 적용 첫 단계는 코덱을 구현하고 인코더로 생성한 바이트열을 golden 파일로 커밋하는 것이며, 이후 클라이언트/서버 양쪽 테스트가 그 파일을 바이트-for-바이트로 왕복 검증해야 한다. 위 블록이 고정하는 것은 오프셋·크기·의미·총 길이 합산이며, 총 길이는 언제나 12 + 본문 필드 합으로 다시 계산해 일치시켜야 한다.

public static class NavGeometry
{
    public static double Distance(double x0, double z0, double x1, double z1)
    { double dx = x1 - x0, dz = z1 - z0; return Math.Sqrt(dx * dx + dz * dz); }

    // 포털 선분을 등 간격으로 나눈 내부/끝점 샘플 좌표를 만든다.
    // ★ 간격은 서버가 결정한다. 클라이언트 좌표·길이는 어떤 경우에도 쓰지 않는다.
    public static int StepsForOverlap(double overlapMeters, double maxSpacing, int minSteps, int maxSteps)
        => (int)Math.Clamp(Math.Ceiling(overlapMeters / Math.Max(1e-6, maxSpacing)), minSteps, maxSteps);

    public static void PointAt(in PortalRef p, double t, out double x, out double z)
    {
        double u = Math.Clamp(t, 0.0, 1.0);
        x = p.A0X + (p.A1X - p.A0X) * u;
        z = p.A0Z + (p.A1Z - p.A0Z) * u;
    }

    // 다각형 내부 "다리(leg)" 선분 — 볼록 가정 아래 다각형 내부에 있으므로 통행 분류는 이미 보장되지만,
    // 오염/비볼록 다각형을 잡아내기 위해 중간 1점을 probe 한다.
    public static void LegMidpoint(in PolygonRef poly, in PortalRef portal, out double x, out double z)
    { x = (poly.CentroidX + portal.MidX) * 0.5; z = (poly.CentroidZ + portal.MidZ) * 0.5; }
}

7.3 본체 의사코드


// AI_Work/PathValidation/PathPlanValidator.cs   (제안, 미생성)
public sealed class PathPlanValidator
{
    private readonly PathPlanValidationContext _ctx;
    private readonly PathPlanValidationPolicy  _p;   // 클램프된 사본

    public PathPlanValidator(PathPlanValidationContext context) { /* 클램프 + 가드 */ }

    public PathPlanValidationResult Validate(PathPlanRequestV1? req)
    {
        try { return ValidateCore(req); }
        catch (Exception ex)   // fail-closed: 예외를 호출자에게 전파하지 않는다
        {
            return Build(planSequence: 0, PathPlanReason.InvalidRequest,
                         detail: "internal:" + ex.GetType().Name, sampled: 0, total: 0, failClosed: true);
        }
    }

    private PathPlanValidationResult ValidateCore(PathPlanRequestV1? req)
    {
        if (req is null) return Reject(PathPlanReason.InvalidRequest, "request_null");

        string fingerprint = PathPlanFingerprint.Compute(req);   // 감사/캐시용, 클라이언트 미전송

        // ── 2) 구조(형식) 검사 — 결정적, 난수 미사용 ─────────────────
        if (!TryStructural(req, out string why)) return Reject(PathPlanReason.InvalidRequest, why);

        // ── 3) 요청 예산 ────────────────────────────────────────────
        if (_ctx.RequestsInCurrentWindow > _p.MaxRequestsPerWindow)
            return Reject(PathPlanReason.RateLimited, "window_budget");

        // ── 4) 순서 / 신선도 ────────────────────────────────────────
        if (req.PlanSequence <= 0) return Reject(PathPlanReason.StaleSequence, "sequence_nonpositive");
        if (_ctx.LastObservedPlanSequence >= 0 && req.PlanSequence <= _ctx.LastObservedPlanSequence)
            return Reject(PathPlanReason.StaleSequence, "sequence_regress");
        if (_ctx.ServerNowUtcMs - (long)req.ClientComputedAtUtcMs > _p.MaxPlanAgeMs)
            return Reject(PathPlanReason.StaleSequence, "plan_age");
        if ((long)req.ClientComputedAtUtcMs - _ctx.ServerNowUtcMs > _p.MaxClientClockSkewMs)
            return Reject(PathPlanReason.InvalidRequest, "client_clock_ahead");

        // ── 5) 시작점 (서버 권위 다각형) ────────────────────────────
        if (_ctx.AuthoritativeShipPolygon >= 0 && _ctx.AuthoritativeShipPolygon != req.StartPolygonIndex)
            return Reject(PathPlanReason.StartMismatch, "start_polygon_mismatch");

        // ── 6) ★ 버전: 인덱스를 하나라도 해석하기 전에 ───────────────
        if (!string.Equals(req.NavMeshVersion, _ctx.NavMesh.Version, StringComparison.Ordinal))
            return Reject(PathPlanReason.ResyncRequired, "version_mismatch");   // ← 아래에서 인덱스를 만지지 않는다

        // ── 7) 인덱스 범위 (이제서야 조회해도 된다) ──────────────────
        int steps = req.PathPolygon.Length - 1;                     // >= 1 (단계 1 이 길이 ≥ 2 를 보장)
        for (int i = 0; i <= steps; i++)
            if (!_ctx.NavMesh.TryGetPolygon(req.PathPolygon[i], out _))
                return Reject(PathPlanReason.PlanInvalid, "path_polygon_out_of_range");
        foreach (var v in req.Visits)
            if (!_ctx.NavMesh.TryGetPolygon(v.PolygonIndex, out _))
                return Reject(PathPlanReason.PlanInvalid, "visit_polygon_out_of_range");

        // ── 8) 통로 구조: 끝점 · 압축 · 인접성 · 포털 모호성 ────────
        if (req.PathPolygon[0]      != req.StartPolygonIndex)      return Reject(PathPlanReason.PlanInvalid, "start_endpoint_mismatch");
        if (req.PathPolygon[steps]  != req.DestinationPolygonIndex) return Reject(PathPlanReason.PlanInvalid, "dest_endpoint_mismatch");

        var portals = new PortalRef[_p.MaxPortalsPerPair][];
        var stepLen = new double[steps];        // 서버 계산 길이(가중밴드용). 클라이언트 값 아님
        var cumEnd  = new double[steps];
        double total = 0.0;

        for (int s = 0; s < steps; s++)
        {
            int a = req.PathPolygon[s], b = req.PathPolygon[s + 1];

            // ★ 압축 규칙: 인접 동일 인덱스는 클라이언트가 압축해야 한다. 그대로면 거절.
            if (a == b) return Reject(PathPlanReason.PlanInvalid, "duplicate_adjacent_polygon");

            var pa = _ctx.NavMesh.TryGetPolygon(a, out var polyA) ? polyA : default;
            var pb = _ctx.NavMesh.TryGetPolygon(b, out var polyB) ? polyB : default;

            var buf = new PortalRef[_p.MaxPortalsPerPair];          // <= 4, 할당 상한 고정
            int nPortal = _ctx.NavMesh.GetPortals(a, b, buf);
            if (nPortal > _p.MaxPortalsPerPair) return Reject(PathPlanReason.PlanInvalid, "portal_ambiguous");
            // ★ 코드 7 의 유일한 발생 지점: 공유 포털 0개 = 비인접 점프.
            if (nPortal <= 0)                 return Reject(PathPlanReason.PathDiscontinuity, "not_adjacent");

            portals[s] = new PortalRef[nPortal];                   // 정확한 개수만 복사
            Array.Copy(buf, portals[s], nPortal);

            double d = NavGeometry.Distance(polyA.CentroidX, polyA.CentroidZ, polyB.CentroidX, polyB.CentroidZ);
            // ★ v1 계약에 길이 상한/하한 검사는 없다. d == 0 은 길이 위반이 아니라 통로 구조
            //   이상(두 다각형 중심점 중복)이므로 PLAN_INVALID(6) 으로 거절한다.
            if (d <= 0.0) return Reject(PathPlanReason.PlanInvalid, "degenerate_step");
            stepLen[s] = d; total += d; cumEnd[s] = total;
        }

        // ── 9) ★ 업프런트 예산 산출 — probe 를 "한 번도" 호출하지 않은 상태 ──
        var budget = new ProbeBudget(_p.MaxGeometryProbesPerValidation);
        int bands = ComputeBandCount(steps);
        var plan = BuildBandPlan(bands, stepLen, cumEnd, total);        // 결정론적
        long requiredProbes = 0;
        for (int k = 0; k < bands; k++)
            requiredProbes += ProbesForStep(plan[k], polyLookup, portals);
        requiredProbes += req.Visits.Count;
        if (requiredProbes > budget.Limit) return Reject(PathPlanReason.RateLimited, "probe_budget_insufficient", 0, steps);
        if ((long)req.Visits.Count * req.PathPolygon.Length > _p.MaxVisitScanComparisons)
            return Reject(PathPlanReason.RateLimited, "visit_scan_budget", 0, steps);

        // ── 10) 무작위 기하 검사 (이하 만 fail-closed) ────────────────
        var outcome = SampleAndProbe(req, plan, portals, fingerprint, budget, out int sampled);
        if (outcome.Reason != PathPlanReason.Valid) return Reject(outcome.Reason, outcome.Detail, sampled, steps);

        // ── 11) 방문 계획 ────────────────────────────────────────────
        var visitOutcome = CheckVisits(req, budget);
        if (visitOutcome.Reason != PathPlanReason.Valid)
            return Reject(visitOutcome.Reason, visitOutcome.Detail, sampled, steps);

        // ── 12) 통과 — budget 이 정확히 0 이고 필요한 probe 를 전부 소모했을 때만 도달 ──
        if (budget.Remaining != 0) return Reject(PathPlanReason.RateLimited, "probe_budget_leak", sampled, steps);
        return Build(req.PlanSequence, PathPlanReason.Valid, "ok", sampled, steps, failClosed: false);
    }
}


7.4 길이 가중 밴드 + 무작위 표본


이전 설계("무작위 스텝 N개 추출")의 두 문제를 그대로 유지한다: (a) 반복 추출은 신발끈 문제로 budget개를 못 맞추면 조용히 적게 검사하고 accepted 가 된다, (b) 필요한 probe 수가 실행 중에만 누적된다. 아래 설계는 커버리지를 결정론적으로 보장하고 필요 probe 수를 probe 호출 전에 확정한다.


// 항로 전체 길이 total 을 bands 개의 "연속 밴드"로 균등 분할한다.
// 밴드 k = [ total*k/bands , total*(k+1)/bands )   (k = 0 … bands-1)
// 각 밴드는 "밴드 시작점까지 누적거리가 밴드 시작을 넘은 첫 스텝"을 담당한다.
//  · bands == steps 이면 밴드 k ↔ 스텝 k 일대일 대응 = 전수 검사.
//  · 그 외에는 각 밴드가 최소 1개 probe 를 갖는다 → 길이 기준 균등 커버리지 보장.
//  · 길이는 서버가 계산한 중심 거리다. 클라이언트 값은 길이 가중에 절대 쓰이지 않는다.
private int ComputeBandCount(int steps)
{
    double target = Math.Max(_p.MinSampledSteps, Math.Ceiling(steps * _p.MinSampleFraction));
    int bands = (int)Math.Clamp(target, 1, Math.Min(_p.MaxSampledSteps, steps));
    return Math.Max(1, bands);       // steps >= 1 이므로 항상 1 이상
}

private readonly struct BandPlan
{
    public int    StepIndex;        // 이 밴드를 담당하는 통로 스텝
    public double StepStartMeters;  // cumEnd[step] - stepLen[step]
    public double StepLengthMeters; // stepLen[step] (> 0)
    public double BandStart;        // 밴드 시작 누적거리
    public double BandEnd;          // 밴드 끝 누적거리
    public double OverlapMeters;    // min(cumEnd, BandEnd) - max(StepStart, BandStart) > 0
}

// ★ 순수 결정론적. 난수를 여기서 쓰지 않아야 9) 의 "업프런트 필요량"을 정확히 산출할 수 있다.
private BandPlan[] BuildBandPlan(int bands, double[] stepLen, double[] cumEnd, double total)
{
    var plan = new BandPlan[bands];
    for (int k = 0; k < bands; k++)
    {
        double bStart = total * k / bands;
        double bEnd   = total * (k + 1) / bands;
        int lo = 0, hi = cumEnd.Length - 1;
        while (lo < hi) { int mid = lo + (hi - lo) / 2; if (cumEnd[mid] > bStart) hi = mid; else lo = mid + 1; }
        int step = lo;
        double stepStart = cumEnd[step] - stepLen[step];
        double overlap   = Math.Min(cumEnd[step], bEnd) - Math.Max(stepStart, bStart);
        if (!(overlap > 0.0)) throw new InvalidOperationException("band overlap must be > 0");  // 방어적
        plan[k] = new BandPlan { StepIndex = step, StepStartMeters = stepStart, StepLengthMeters = stepLen[step],
                                 BandStart = bStart, BandEnd = bEnd, OverlapMeters = overlap };
    }
    return plan;
}

// 밴드가 담당하는 스텝의 모든 공유 포털에 필요한 probe 수 (9) 에서 쓰이는 것과 정확히 같은 수를 돌려준다.
private long ProbesForStep(in BandPlan b, PortalRef[][] portals, IReadOnlyDictionary<int, PolygonRef> polys)
{
    int step = b.StepIndex;
    long n = 0;
    foreach (var portal in portals[step])
    {
        int steps_ = NavGeometry.StepsForOverlap(portal.LengthMeters, _p.MaxSampleSpacingMeters,
                                                 _p.MinStepsPerPortal, _p.MaxStepsPerPortal);
        n += _p.PortalEndpointsProbes + steps_;                 // 양끝점 + 내부
        n += _p.LegProbesPerStep * 2;                           // 두 다각형 내부 다리 중간점
    }
    return n;
}


// 7.6.3 서버Controlled 난수원
private sealed class CsprngSource : IRandomSource   // 기본 경로
{
    private readonly RandomNumberGenerator _rng = RandomNumberGenerator.Create();
    public double NextUnitDouble()                      // [0, 1)
    {
        Span<byte> b = stackalloc byte[8];
        _rng.GetBytes(b);
        return (BitConverter.ToUInt64(b) >> 11) * (1.0 / 9007199254740992.0);   // 53비트만 → [0,1)
    }
}

private sealed class HmacSource : IRandomSource      // 감사 가능 경로(선택)
{
    // 스트림 = HMAC-SHA256(serverSecret, nonce ‖ serverRequestContext ‖ planFingerprint ‖ counter)
    // nonce 는 요청마다 CSPRNG 로 새로 뽑고 서버 로그에만 남긴다(클라이언트 미전송).
}
// 난수원이 서버 측에만 존재하고, 클라이언트가 보낼 수 있는 seed/표본인덱스/힌트 필드는 wire 본문에 없다.


// 표본 probe 루프 — 반복 횟수 = 밴드 수 = 계획에 이미 고정. 무한/과도 루프가 구조적으로 불가능하다.
private ValidationStepResult SampleAndProbe(PathPlanRequestV1 req, BandPlan[] plan, PortalRef[][] portals,
                                             string fingerprint, ProbeBudget budget, out int sampled)
{
    using var rng = CreateRandomSource(fingerprint, out string? auditNonce);   // CSPRNG 또는 HMAC
    var probedSteps = new HashSet<int>();
    sampled = 0;

    for (int k = 0; k < plan.Length; k++)
    {
        var b = plan[k];
        int s = b.StepIndex;
        probedSteps.Add(s);

        var stepPortals = portals[s];

        // ★ 스텝이 공유 포털을 여러 개 가지면 전부 검사한다. 클라이언트가 어떤 포털을 지났다고
        //   말하든, 서버는 통과 가능한 모든 통로를 확인해야 "좁은 포털 타협"을 막을 수 있다.
        foreach (var portal in stepPortals)
        {
            int steps_ = NavGeometry.StepsForOverlap(portal.LengthMeters, _p.MaxSampleSpacingMeters,
                                                     _p.MinStepsPerPortal, _p.MaxStepsPerPortal);

            // (a) 양끝점 + 내부 steps_ 개  → steps_ + 2 probe (7) 산출과 정확히 같은 수
            for (int i = 0; i <= steps_ + 1; i++)
            {
                double t = (double)i / (steps_ + 1);
                NavGeometry.PointAt(portal, t, out double x, out double z);
                if (!ProbeOnce(x, z, budget)) return ValidationStepResult.Fail(PathPlanReason.RateLimited, "probe_budget_exhausted");
            }

            // (b) 다각형 내부 "다리" 중간점 — 볼록 가정 위반/오염 다각형 방어
            _ctx.NavMesh.TryGetPolygon(req.PathPolygon[s],     out var pa);
            _ctx.NavMesh.TryGetPolygon(req.PathPolygon[s + 1], out var pb);
            NavGeometry.LegMidpoint(pa, portal, out double lx0, out double lz0);
            NavGeometry.LegMidpoint(pb, portal, out double lx1, out double lz1);
            if (!ProbeOnce(lx0, lz0, budget)) return ValidationStepResult.Fail(PathPlanReason.RateLimited, "probe_budget_exhausted");
            if (!ProbeOnce(lx1, lz1, budget)) return ValidationStepResult.Fail(PathPlanReason.RateLimited, "probe_budget_exhausted");
        }
    }

    sampled = probedSteps.Count;   // 응답 sampledSegments = 실제 검사한 서로 다른 스텝 수
    return ValidationStepResult.Ok();
}

// ★ 예산은 probe 호출 직전에만 소모한다. 순서가 뒤집히면 navmesh 를 읽은 뒤에야 거절된다.
private bool ProbeOnce(double x, double z, ProbeBudget budget)
{
    if (!budget.TryConsume()) return false;
    NavSample r;
    try { r = _ctx.NavMesh.Classify(x, z); }
    catch { r = default; }   // 권한 계층 오류 → fail-closed

    if (r.Classification == NavClassification.Forbidden) { _lastFail = PathPlanReason.ForbiddenRegion; return false; }
    if (r.Classification != NavClassification.Navigable)  { _lastFail = PathPlanReason.OffMeshSample;     return false; }
    if (!r.HasUsableClearance || r.ClearanceMeters < _p.RequiredClearanceMeters)
                                                     { _lastFail = PathPlanReason.ForbiddenRegion;   return false; }
    return true;
}

// 내부 단계 결과. 응답 모델이 아니며 직렬화되지 않는다.
internal readonly struct ValidationStepResult
{
    public PathPlanReason Reason { get; init; }
    public string         Detail { get; init; } = "ok";
    public static ValidationStepResult Ok()    => new() { Reason = PathPlanReason.Valid };
    public static ValidationStepResult Fail(PathPlanReason r, string d) => new() { Reason = r, Detail = d };
}

// 검증 1회(=요청 1회) 전용 카운터. 이 객체는 해당 검증 스레드만 접근한다.
internal sealed class ProbeBudget
{
    public int Limit { get; }
    private int _used;
    public int Used      => _used;
    public int Remaining => Limit - _used;
    public ProbeBudget(int limit) => Limit = Math.Max(1, limit);
    // ★ 반드시 _ctx.NavMesh.Classify(...) 호출 "직전"에 호출한다. probe 가 끝난 뒤 세면 안 된다.
    public bool TryConsume() { if (_used >= Limit) return false; _used++; return true; }
}


7.5 방문 계획 — 인덱스 기반 순서 판정, probe 는 예산 이후


private ValidationStepResult CheckVisits(PathPlanRequestV1 req, ProbeBudget budget)
{
    var visits = req.Visits;

    // 결정론적 스캔 예산 (probe 를 쓰지 않는 작업도 유한해야 한다).
    if ((long)visits.Count * req.PathPolygon.Length > _p.MaxVisitScanComparisons)
        return ValidationStepResult.Fail(PathPlanReason.RateLimited, "visit_scan_budget");

    // 다각형 통로에서는 "거리 반경"이 아니라 포함 여부로 도달을 정의한다 (정수 비교, 부동소수점 없음).
    var firstStep = new int[visits.Count];
    for (int v = 0; v < visits.Count; v++)
    {
        firstStep[v] = -1;
        for (int i = 0; i <= req.PathPolygon.Length - 1; i++)   // 마지막 다각형 포함
            if (req.PathPolygon[i] == visits[v].PolygonIndex) { firstStep[v] = i; break; }

        if (firstStep[v] < 0)   // 통로에 visit 다각형이 없음
            return ValidationStepResult.Fail(PathPlanReason.VisitUnreached, "visit_not_on_path");

        // visit 다각형 자체가 통행 가능한지 (중복 다각형이면 probe 를 아끼기 위해 캐시)
        if (!_ctx.NavMesh.TryGetPolygon(visits[v].PolygonIndex, out var poly)) 
            return ValidationStepResult.Fail(PathPlanReason.VisitUnreached, "visit_polygon_missing");
        NavSample vs = _ctx.NavMesh.Classify(poly.CentroidX, poly.CentroidZ);
        if (vs.Classification != NavClassification.Navigable)
            return ValidationStepResult.Fail(PathPlanReason.VisitUnreached, "visit_polygon_off_mesh");
        if (!budget.TryConsume())   // ★ probe 전에 예산 소모
            return ValidationStepResult.Fail(PathPlanReason.RateLimited, "probe_budget_exhausted");
    }

    // 순서 판정은 정수 인덱스 비교 → 부동소수점 동률/오차 모호성이 없다.
    for (int v = 1; v < visits.Count; v++)
        if (firstStep[v] < firstStep[v - 1])      // 같은 다각형(동률)은 불일치가 아니다
            return ValidationStepResult.Fail(PathPlanReason.VisitOrderMismatch, "visit_order");

    return ValidationStepResult.Ok();
}



순서 규칙은 단계 2(구조)에서 한 번만 확인한다. visits[k].order == k는 요청 본문의 배열 순서 자체가 선언 순서이므로, 단계 2에서 어긋나면 INVALID_REQUEST. 단계 11의 VISIT_ORDER_MISMATCH는 "선언 순서"가 아니라 "선언된 순서가 실제 통로 통과 순서와 다르다"는 별개의 실패다. 두 실패를 같은 코드로 뭉개지 않는다.


serverPlanId는 매 응답마다 CSPRNG 16바이트에서 생성한 불투명 값(sp1_ + 소문자 hex)이며 요청 내용에서 파생하지 않는다. accepted가 아니면 본문에는 길이 0(null)로 인코딩한다. 표본 스텝·포털·감사 nonce는 응답 본문에 넣지 않고 서버 로그/DB에만 남긴다.





7.1 좌표계 규약 (server global ↔ navmesh) — 확정/미확정 분리


이 절은 서버 권위 좌표와 클라이언트 navmesh 좌표의 관계를 확정된 것과 미확정인 것으로 나눠 기록한다. 검증 로직은 다각형 인덱스만 다루므로 어떤 좌표도 wire 에 실리지 않는다(7장 클라이언트 매핑). 좌표는 서버 navmesh 내부에만 존재한다.


    

  • 확정 — 서버 글로벌 좌표. NWO_RegionNode/MoveMent.cs의 MoveMent.globalPosition은 nwo_Vector3(Int32 X, Int32 Y, Int32 Z) 기반의 대형 정수 XYZ 좌표다. 실사용 범위/정밀도는 저장소에 정의되어 있지 않다.
    • 

  • 확정 — 타일 나누기. 같은 생성자에서 tilePosition = (X/2560, Z/2560), position = (X/2560, Y, Z/2560)로 X/Z 를 정수 2560 으로 나눈다. 즉 서버가 사용하는 타일 좌표 단위는 2560 좌표 단위(tile divisor 2560)이며, 이 2560이 서버 측에서 확정된 유일한 수치다. 2560 의 물리적 의미(미터/센티미터/기타)는 저장소에 명시가 없다.
    • 

  • 확정 — EchoServerHandler 잔여 스케일. 같은 핸들러에서 타일 연산 후 좌표에 ×10을 곱해 전달한다(position * 10). 이 10배 스케일이 서버 측 파이프라인에서 확정된 마지막 승수다.
    • 

  • 미확정 — 대형 월드 전체의 스케일 크기. 서버 저장소에서 확정된 것은 타일 나누기 수치 2560 과 그 좌표 단위뿐이다. 월드의 총 길이나 전역 축척(world scale)을 이 2560으로부터 환산해 확정하지 않는다. 알려진 것은 2560 좌표 단위까지다.
    • 

  • 확정(구조) — 클라이언트 변환 표면. 클라이언트 ShipAI는 GeoNavMapTransform/NavMeshJsonRenderer 를 통해 스케일 적응된 대형 좌표를 렌더링하고, 변환 파라미터 centerPivot, worldScale, invertZ 는 구성 가능(configurable)하게 노출된다. navmesh JSON(format navmesh-merged-convex, 정점 [x, y] with y down, polygons/portals/centroids 포함)이 선언하는 메타데이터는 입력 계약일 뿐 서버-클라이언트 스케일 대응을 증명하지 않는다.
    • 

  • 미확정 — 원점·축/부호·스케일 대응. 서버 글로벌 X/Y/Z 와 navmesh 로컬 좌표 사이의 원점(origin), 축 매핑과 부호(invertZ, y-down vs Z-up), 스케일 비율은 서버 저장소 코드에서 확정되어 있지 않다(클라이언트 애셋은 서버 저장소 밖의 별도 체크아웃에 있다). 확정하려면 실제 클라이언트 변환 소스(GeoNavMapTransform/NavMeshJsonRenderer의 실제 centerPivot/worldScale/invertZ 값)와 서버 좌표 소스(MoveMent.globalPosition 및 실제 클라이언트 송신 좌표)를 대조 캘리브레이션해야 한다. 그 전까지는 미해결/구성 가능 항목으로 취급한다.
    • 

  • 비권위 가정 금지. navmesh JSON 이나 화면상의 격자/타일 개수로 worldScale 을 역산해 확정하지 않는다. 서버-클라이언트 좌표 대응은 반드시 명시적 구성/캘리브레이션 파라미터로 다뤄야 하며, 저장소 코드가 이를 정의하지 않는 한 특정 수치를 확정하지 않는다.
    • 

  • 권위 소스. navmesh.json 은 서버가 다각형/포털 기하를 읽어들이는 소스이며, 서버가 그 기하를 소유한다. 클라이언트는 통과한 다각형 인덱스만 순서대로 보내며 기하를 전송하지 않는다. 클라이언트가 보낸 어떤 수치도 좌표 근거로 사용하지 않는다.
    • 



좌표 변환(서버 글로벌 XZ ↔ navmesh 로컬)의 비율·원점·축 부호는 실제 클라이언트 변환(GeoNavMapTransform/NavMeshJsonRenderer)과 서버 좌표 소스(MoveMent.cs)를 기준으로 캘리브레이션해 확정한다. 확정 전까지 IAuthoritativeNavMesh의 Classify/LocatePolygon 에 들어가는 nav 좌표의 절대 스케일은 미해결이며, 검증 로직은 nav 좌표 한계 안에서 상대적으로만 동작한다(비율·스텝 길이·통행 폭 임계값은 navmesh 빌드와 캘리브레이션 확정 후에 확정한다).

8. DotNetty 통합 — 동기 ChannelRead vs 오프루프 검증

먼저 규칙을 고정한다. DotNetty의 ChannelRead/ByteToMessageDecoder.Decode 는 채널 이벤트 루프에서 동기 호출된다. 이 안에서 await, .Result, .Wait(), 스레드 슬립은 그 채널(및 같은 워커 스레드의 다른 채널)을 정지시키므로 금지한다. 반대로 ByteBuf 는 이벤트 루프가 반환한 뒤 곧 해제(release) 되기 때문에 비동기 continuation 안에서 절대 참조하면 use-after-release 이다. 해법은 "상한이 있는 바이트를 먼저 복사 → 복사본으로 오프루프 검증 → 응답은 이벤트 루프에 기록" 이다.

8.1 기존 0x02/0x04 와의 프레이밍 호환성 (실측 기반)

현재 Program.cs의 19100 파이프라인은 pipeline.AddLast("echo", new EchoServerHandler()) 한 줄이고 길이 프레이밍 decoder가 없다. EchoServerHandler 는 ChannelHandlerAdapter 를 직접 상속해 buffer.GetByte(0), GetByte(1) 로 (opcode, 서브타입) 을 분기하고, readerIndex 를 전진시키지 않으며 수신 버퍼를 해제하지 않는다. 즉 기존 프로토콜은 길이 프리픽스 없이 고정 오프셋에 의존한다.

opcode서브타입의미핸들러가 읽는 최대 오프셋필요 최소 길이
0x020x00위치정보 갱신요청(응답 송신)56
0x020x01위치정보 동기화2425
0x020x02관성항법 동기화1112
0x040x00오브젝트 위치정보 동기화1819
0x040x01오브젝트 위치정보 동기화(2)2021
0x040x02오브젝트 관성항법 동기화1011
0x040x03빈 패킷12
0x040x04공격(피격 선분)2728
좋은 소식. 새 경로 프로토콜은 기존 0x04 opcode를 공유하며 subtype 0x10(요청)/0x11(응답)으로 분기한다. 따라서 buffer[0]가 0x04일 때 buffer[1](subtype)을 peeking(인덱스 미전진)하여 0x02/0x04 레거시와 0x04/0x10·0x04/0x11 경로 프로토콜을 안전하게 판별할 수 있다. opcode만으로는 두 경로를 가를 수 없고 (opcode, subtype) 두 바이트가 필요하다.
위험: 같은 TCP 연결에서 두 프로토콜을 섞어 읽는 것은 안전하지 않다. EchoServerHandler 는 readerIndex 를 전진시키지 않고 첫 바이트 절대 오프셋만 읽는다. 따라서 앞에 어떤 분할기가 들어가든 그 핸들러가 보는 바이트 열이 달라진다(오늘은 단일 terminal handler 라서 우연히 동작하는 상태다). 또한 그 핸들러는 버퍼를 소비하지도 해제하지도 않으므로, 한 연결에서 뒤따르는 0x04/0x10·0x04/0x11 패킷은 사실상 도달 불가/오독된다.

8.2 선택지 A (권장) — 공유 19100 포트 — (opcode, subtype) 기반 혼합 디스패처

기존 19100 파이프라인을 한 줄도 바꾸지 않고 19100 (공유 포트) 에 경로 프로토콜 전용 파이프라인을 만든다. 이 선택지가 유일하게 기존 동작에 위험을 추가하지 않는다.

// Program.cs RunServerAsync 에 추가할 두 번째 bootstrap (제안, 미적용)
// 기존 19100 bootstrap 은 그대로 두고, 19100 (공유 포트) 만 경로 프로토콜 전용으로 만든다.
var pathPlanBoss   = new MultithreadEventLoopGroup(1);
var pathPlanWorker = new MultithreadEventLoopGroup();
var pathPlanBootstrap = new ServerBootstrap();
pathPlanBootstrap
    .Group(pathPlanBoss, pathPlanWorker)
    .Channel<TcpServerSocketChannel>()
    .Option(ChannelOption.SoBacklog, 100)
    .Option(ChannelOption.TcpNodelay, true)
    .ChildHandler(new ActionChannelInitializer<ISocketChannel>(channel =>
    {
        IChannelPipeline pipeline = channel.Pipeline;
        pipeline.AddLast("pathplan.frame", new PathPlanFrameDecoder());          // 동기, await 금지
        pipeline.AddLast("pathplan.handler", new PathPlanValidationHandler(CreateValidationContext));

    }));
IChannel pathPlanChannel = await pathPlanBootstrap.BindAsync(19100 (공유 포트));

// RequestsInCurrentWindow, AuthoritativeShipPolygon, IAuthoritativeNavMesh 구현체를 여기서 주입한다.
// 이 리스너의 인증은 19100 세션과 별개로 建立되므로, 인증 세션 없이는 요청을 받아들이지 않는다

8.3 선택지 B — 같은 포트 혼합 디스패처 (조건부, 비권장)

같은 포트가 불가피한 경우에만 검토한다. 아래는 안전 규칙 전체이며, 마지막 항목 때문에 기본값이 아니다.

// AI_Work/PathProtocol/MixedProtocolDispatcher.cs   (제안, 미생성)
// "echo" 핸들러 앞에 놓는다. 두 프로토콜을 한 누적 버퍼에서 순서대로 잘라내는 유일한 방법.
protected override void Decode(IChannelHandlerContext ctx, IByteBuffer input, List<object> output)
{
    for (;;)
    {
        // (0) ★ 최소 1바이트: TCP 는 메시지 경계를 보장하지 않는다. opcode 바이트조차 없을 수 있다.
        if (input.ReadableBytes() < 1) return;
        byte op = input.GetByte(input.ReaderIndex);        // ★ peek. readerIndex 를 움직이지 않는다.

        if (op == PathPlanContract.OpcodePathPlan)
        {
            // ── 0x04/0x10·0x04/0x11 경로: 자기경계 헤더 ──────────────────────────────
            if (input.ReadableBytes() < PathPlanContract.HeaderLength) return;   // 헤더 대기
            if (!PathPlanHeaderCodec.TryRead(input.GetReaderSlice(), out var h, out var err)) return EmitFrameError(ctx, err);

            // 2) 상한 검증 — 메모리를 제한하는 유일한 지점. 할당보다 먼저.
            if (h.TotalPacketLength < PathPlanContract.MinPacketLength || h.TotalPacketLength > PathPlanContract.MaxPacketLength)
                throw new CorruptedFrameException("totalPacketLength out of range: " + h.TotalPacketLength);

            if (input.ReadableBytes() < (int)h.TotalPacketLength) return;      // 본문 대기

            byte[] raw = new byte[h.TotalPacketLength];                           // 상한 통과 후 NOW 복사
            input.ReadBytes(raw);                                                 // 정확히 totalPacketLength 만큼만 소비
            output.Add(PathPlanFrame.Parse(raw));                                 // ByteBuf 캡처 금지
            continue;
        }

        // ── 기존 0x02/0x04 경로: 길이 프리픽스 없음 → 고정 길이 테이블로만 경계를 잡는다 ──
        if (op is 0x02 or 0x04)
        {
            if (input.ReadableBytes() < 2) return;                              // 서브타입 대기
            byte sub = input.GetByte(input.ReaderIndex + 1);
            if (!LegacyPacketLength.TryGet(op, sub, out int fixedLen))
            {
                // ★ 길이를 추측하지 않는다. 모르는 조합은 fail-closed 로 종료한다.
                throw new CorruptedFrameException($"unknown legacy packet 0x{op:X2}/0x{sub:X2}");
            }
            if (input.ReadableBytes() < fixedLen) return;                       // 고정 길이 대기

            byte[] slice = new byte[fixedLen];
            input.ReadBytes(slice);                                               // 정확히 fixedLen 만큼만 소비
            output.Add(Unpooled.WrappedBuffer(slice));                            // "echo" 는 IByteBuffer 를 기대
            continue;
        }

        // ★ 알 수 없는 첫 바이트: 1바이트 건너뛰고 재탐색하지 않는다(패킷 중간에서 동기화되어
        //   잘못된 경계를 만들 수 있다). 연결을 종료해 상태를 명확히 한다.
        throw new CorruptedFrameException("unknown leading byte: 0x" + op.ToString("X2"));
    }
}

internal static class LegacyPacketLength
{
    // EchoServerHandler.ChannelRead 가 실제로 읽는 최대 오프셋 + 1 (소스 기준, 미검증 가정).
    // ★ 이 테이블은 기존 동작의 관찰값이지 계약이 아니다. 클라이언트가 더 긴 패킷을
    //   보낼 가능성까지 배제하는 근거가 없으므로, 통합 시 클라이언트 송신 코드와 대조 검증이 선행되어야 한다.
    private static readonly (byte Op, byte Sub, int Len)[] Table =
    {
        (0x02, 0x00,  6), (0x02, 0x01, 25), (0x02, 0x02, 12),
        (0x04, 0x00, 19), (0x04, 0x01, 21), (0x04, 0x02, 11),
        (0x04, 0x03,  2), (0x04, 0x04, 28),
    };
    public static bool TryGet(byte op, byte sub, out int len)
    {
        foreach (var t in Table) if (t.Op == op && t.Sub == sub) { len = t.Len; return true; }
        len = 0; return false;
    }
}
선택지 B 의 전제 조건 (기본값이 아닌 이유). 디스패처가 "고정 길이만큼 소비하고 정확한 슬라이스를 넘긴다"는 것은, downstream EchoServerHandler 가 그 슬라이스를 0번 오프셋부터 읽는다는 가정 위에 성립한다. 현재 핸들러는 readerIndex 를 무시하고 절대 오프셋만 읽으므로 이 가정과 우연히 맞지만, 프레임마다 정확히 한 패킷만 전달된다는 보장은 없다(버퍼 미해제·미소비). 즉 선택지 B 는 EchoServerHandler 를 함께 손봐야(SimpleChannelInboundHandler 전환 또는 슬라이스 소비 규칙 확립) 완전해진다. 그 수정은 이번 범위 밖이며, 그래서 8.2(별도 포트)가 기본값이다.

8.4 0x04/0x10 프레이밍 디코더 — 동기, 상한 복사, ByteBuf 비노출 (공유 포트 디스패처)

// AI_Work/PathProtocol/PathPlanFrameDecoder.cs   (제안, 미생성)
public sealed class PathPlanFrameDecoder : ByteToMessageDecoder
{
    // ★ Decode 는 채널 이벤트 루프에서 동기 호출된다. await/block 금지.
    protected override void Decode(IChannelHandlerContext context, IByteBuffer input, List<object> output)
    {
        // (1) opcode 바이트조차 없으면 아무것도 소비하지 않고 반환 → 다음 read 에서 이어진다.
        if (input.ReadableBytes() < 1) return;

        // (2) ★ peek. readerIndex 를 움직이지 않는다 (Dispatcher 분기와 동일한 원칙).
        byte op = input.GetByte(input.ReaderIndex);
        if (op != PathPlanContract.OpcodePathPlan)
        {
            // 경로 프로토콜 파이프라인에 다른 opcode 가 왔다 = 계약 위반(framing 불가).
            throw new CorruptedFrameException("unexpected opcode on path-plan pipeline: 0x" + op.ToString("X2"));
        }

        // (3) 헤더 12바이트 대기. DotNetty 의 Get* 기본은 BE 이므로 헤더는 위 Codec(LE)을 쓴다.
        if (input.ReadableBytes() < PathPlanContract.HeaderLength) return;
        if (!PathPlanHeaderCodec.TryRead(input.GetReaderSlice(), out var h, out var err))
        {
            if (err == PathPlanDecodeError.Truncated) return;   // 대기
            throw new CorruptedFrameException("bad path-plan header: " + err);
        }

        // (4) 상한 검증. 이 검사가 메모리를 제한하는 유일한 지점이며, 할당보다 먼저 온다.
        if (h.TotalPacketLength < PathPlanContract.MinPacketLength || h.TotalPacketLength > PathPlanContract.MaxPacketLength)
            throw new CorruptedFrameException("totalPacketLength out of range: " + h.TotalPacketLength);

        // (5) v1 은 요청만 받는다. 응답(0x11) 을 서버 포트에서 받는 것은 프로토콜 위반.
        if (h.Subtype != PathPlanContract.SubtypeRequest)
            throw new CorruptedFrameException("expected request subtype");

        // (6) 아직 안 모였으면 소비하지 않고 반환. 누적 버퍼 상한은 MaxPacketLength 로 제한된다.
        if (input.ReadableBytes() < (int)h.TotalPacketLength) return;

        // (7) 상한을 통과했으므로 NOW 복사한다. 이 byte[] 가 비동기 경계를 넘길 유일한 값이다.
        byte[] raw = new byte[h.TotalPacketLength];
        input.ReadBytes(raw);                                   // 정확히 totalPacketLength 만큼만 소비

        // (8) 본문 길이는 totalPacketLength - 12 로 유일하게 결정된다. 별도 bodyLength 검증이 필요 없다.
        if (raw.Length - PathPlanContract.HeaderLength < PathPlanContract.MinRequestBodyBytes)
            throw new CorruptedFrameException("request body shorter than contract minimum");

        // (9) 불변 객체로 감싸 파이프라인 아래로 내보낸다. IByteBuffer 를 캡처하지 않는다.
        output.Add(PathPlanFrame.Parse(raw));                   // throws → ExceptionCaught → 채널 닫힘(ByteToMessageDecoder 가 버퍼 해제)
    }
    // 파싱 실패 시에도 누적 버퍼가 해제되므로 예외 전파가 새 leak 이 되지 않는다.
    // 길이/버전 위반은 "조용히 버리지 않는다": 채널을 닫아 반쪽짜리 프레임 상태로 남지 않게 한다.
    // ★ TCP 는 느린 공격자가 totalPacketLength 바이트를 보내놓고 멈출 수 있으므로,
    //   채널별 누적 대기 deadline(예 5s)과 최대 누적 바이트 상한을 두는 것을 권장한다.
}

// AI_Work/PathProtocol/PathPlanFrame.cs   (제안, 미생성)
// ★ 완전히 불변 + ByteBuf 비보유. 이렇게 해야 ChannelRead 를 지난 뒤에도 안전하다.
public sealed class PathPlanFrame
{
    public byte[] Bytes { get; }                              // 헤더+본문 전체 복사본
    public ReadOnlySpan<byte> Body => Bytes.AsSpan(PathPlanContract.HeaderLength);

    public static PathPlanFrame Parse(byte[] raw)
    {
        if (!PathPlanHeaderCodec.TryRead(raw, out var h, out var err)) throw new CorruptedFrameException(err.ToString());
        if (h.TotalPacketLength != (uint)raw.Length) throw new CorruptedFrameException("declared length != actual");
        return new PathPlanFrame(raw);
    }
    // Parse 는 입력 byte[] 를 수정하지 않는다 → 호출자가 가진 프레임은 그대로 불변이다.

8.5 검증 핸들러 — 채널별 순서 보장 체인 + 이벤트 루프 응답 기록

// AI_Work/PathProtocol/PathPlanValidationHandler.cs   (제안, 미생성)
public sealed class PathPlanValidationHandler : ChannelInboundHandlerAdapter
{
    private const int MaxPendingPerChannel = 8;      // 이 값을 넘는 큐 적체는 허용하지 않는다

    private readonly Func<PathPlanFrame, PathPlanValidationContext> _contextFactory;

    // ── 아래 세 필드는 "채널 이벤트 루프 스레드 전용"이다(lock 없이 접근).
    private int  _pending;
    private long _nextArrival;
    private Task _tail = Task.CompletedTask;
    private readonly object _chainGate = new();

    public override void ChannelRead(IChannelHandlerContext ctx, object message)
    {
        if (message is not PathPlanFrame frame) { ctx.FireChannelRead(message); return; }

        // (A) 큐 상한 검사 — 순수한 정수 비교이므로 여기서 즉시 판정해 응답할 수 있다.
        if (_pending >= MaxPendingPerChannel)
        {
            // 디코딩·검증 없이 즉시 limited. 호출자는 지수 백오프로 물러난다.
            WriteOnEventLoop(ctx.Channel, EncodeLimited(frame, planSequence: 0));
            return;
        }
        _pending++;
        long arrival = ++_nextArrival;

        // (B) 순서 보장 체인에 붙인다. 채널마다 1개씩만 "실행 중"이므로 응답이 뒤집히지 않는다.
        //     frame 은 이미 byte[] 복사본이라 채널 상태와 무관하게 안전하다.
        lock (_chainGate)
        {
            _tail = _tail.ContinueWith(_ => HandleAsync(ctx.Channel, frame, arrival),
                                       CancellationToken.None, TaskContinuationOptions.None, TaskScheduler.Default);
        }
    }

    private async Task HandleAsync(IChannel channel, PathPlanFrame frame, long arrival)
    {
        byte[] response;
        try
        {
            // ★ 오프루프. 바이트 바인딩 + 검증 + navmesh 조회가 전부 여기(워커 스레드)에서 일어난다.
            var ctx = _contextFactory(frame);
            PathPlanRequestV1 req;
            PathPlanDecodeError derr;
            if (!PathPlanRequestDecoder.TryDecode(frame.Body, out req, out derr))
            {
                // ★ 디코드 실패는 planSequence 를 알 수 없으므로 0 으로 보낸다. 원인은 서버 로그에만.
                response = PathPlanResponseEncoder.Encode(
                    new PathPlanValidationV1(0, PathPlanDecisionMap.Of(PathPlanReason.InvalidRequest),
                                            PathPlanReason.InvalidRequest, ctx.NavMesh.Version, 0, 0, null),
                    frame.Header.RequestId);
            }
            else
            {
                var result = new PathPlanValidator(ctx).Validate(req);
                response = PathPlanResponseEncoder.Encode(
                    new PathPlanValidationV1(result.PlanSequence, PathPlanDecisionMap.Of(result.Reason),
                                            result.Reason, ctx.NavMesh.Version,
                                            result.SampledSegments, result.TotalSegments, result.ServerPlanId),
                    frame.Header.RequestId);
            }
        }
        catch (Exception ex)
        {
            // fail-closed: 예외 종류는 서버 로그에만, 클라이언트에는 INVALID_REQUEST(1)/rejected(1) 만.
            response = PathPlanResponseEncoder.Encode(
                new PathPlanValidationV1(0, PathPlanDecision.Rejected, PathPlanReason.InvalidRequest,
                                        _fallbackVersion, 0, 0, null),
                frame.Header.RequestId);
        }
        finally
        {
            Interlocked.Decrement(ref _pending);        // 어떤 경로에서도 큐 슬롯은 반드시 반환된다
        }

        // ★ 응답 기록은 반드시 채널 이벤트 루프 위에서. 어느 스레드에서 호출해도 안전하다.
        WriteOnEventLoop(channel, response);
    }

    // ── 응답 기록 ────────────────────────────────────────────────────────────
    private static void WriteOnEventLoop(IChannel channel, byte[] response)
    {
        if (!channel.IsActive) return;                  // 닫힌 채널에 버퍼를 만들지 않는다
        IEventLoop loop = channel.EventLoop;
        if (loop.InEventLoop) WriteUnsafe(channel, response);
        else                  loop.Execute(() => WriteUnsafe(channel, response));
    }

    private static void WriteUnsafe(IChannel channel, byte[] response)
    {
        if (!channel.IsActive) return;
        IByteBuffer buf = Unpooled.CopiedBuffer(response);
        // DotNetty 계약: WriteAndFlushAsync 가 정상 반환되면 버퍼 소유권이 파이프라인으로 이전되고,
        // 호출이 예외를 던지면 이미 해제되어 있다. 따라서 "반환 후 직접 해제"를 하지 않는다.
        _ = channel.WriteAndFlushAsync(buf);
    }
}
이 설계가 지키는 불변식
1) use-after-release 없음. 비동기 경계를 넘는 값은 전부 byte[] 복사본이다. IByteBuffer 는 디코더 밖으로 나가지 않는다.
2) 이벤트 루프 비차단. 본문 바인딩(최대 34,988바이트 계약)과 navmesh 조회(probe 최대 MaxGeometryProbesPerValidation)는 전부 오프루프다.
3) 채널별 순서. 단일 소비자 체인 + 이벤트 루프 기록 → 요청 i 의 응답이 요청 j(i<j) 의 응답보다 늦게 나가지 않는다. 여러 스레드에서 동시에 WriteAndFlushAsync 를 부르면 이 보장이 없어지므로 반드시 WriteOnEventLoop 를 통과시킨다.
4) 응답은 항상 1:1. 상한 초과·디코드 실패·검증 실패·정상 모두 같은 경로로 응답을 쓴다(클라이언트가 타임아웃으로 죽지 않는다).
5) 큐 유계. MaxPendingPerChannel 이 없으면 느린 검증이 무한 체인을 만들어 메모리를 먹는다.
6) 프레임 손상은 응답이 아니라 연결 종료다. 길이/버전 위반은 INVALID_REQUEST 응답을 보내지 않는다 — 손상 프레임에 대한 판정은 신뢰성을 낮춘다.
동기 대안(권장하지 않음). 만약 설계상 디코딩을 반드시 이벤트 루프에서 동기 수행해야 한다면(예: 측정 결과 34,988바이트 바인딩이 예산 안에 들어옴), 위 체인/이벤트루프 기록 부분을 제거하고 핸들러 안에서 그대로 처리한다. 그 경우에도 패킷 상한(1 MiB)·계약 상한·probe 예산은 그대로 유지되어야 하며, 이벤트 루프 점유 시간을 실측해 임계값을 정해야 한다.

9. 미해결 쟁점 / 리스크

  • navmesh 부재. IAuthoritativeNavMesh 를 실제로 구현할 authoritative 소스가 아직 없다. 본 로직은 계약만 제공한다.
  • 다각형 인덱스 규칙 미확정. "서버 navmesh 배열 순번"을 어떻게 결정론적으로 매길지(로드 순서/정렬/파일 저장 규칙) 가 단계 0 정책 결정이다. 규칙이 흔들리면 인덱스가 조용히 무효화되므로 버전 문자열과 함께 동결되어야 한다.
  • 비볼록 다각형. 인접성·다리 검사가 "다각형이 볼록"이라는 전제 위에서 다각형 내부를 안전하다고 본다. 비볼록 다각형이 섞이면 다리 probe 가 일부만 커버한다. navmesh 빌드 단계에서 볼록성을 강제해야 한다.
  • 공유 포털 다중화. MaxPortalsPerPair=4 를 넘는 인접 쌍은 현재 PLAN_INVALID 로 거절한다. navmesh 생성 설정을 조정해 이 상한을 만족시켜야 하며, 그렇지 않으면 정상 항로가 거절된다.
  • 위험 가중 샘플링 미구현. 해안선/협수로/방향변화 위험 가중은 서버 위험 쿼리가 필요해 길이 가중만 구현한다. 추가는 별도 과제.
  • plan 해시 캐시/재요청 비용 제한도 규약 6장에 있으나 상태 저장소가 없어 이번 범위 밖이다(PathPlanFingerprint만 계산).
  • 한계값은 시작점이며 안전 보증값이 아니다. 선박 빔, map 해상도, 업데이트 빈도, 서버 부하 측정 후 확정 필요.
  • nav 좌표계(단위, 축 방향, origin), navmesh 경계/seam, 포털 경계 부동소수점 오차는 단계 0 결정 사항이다. 직전 설계의 WGS84 datum/경도 wrap 문제는 경로 표현에서 위경도를 없애면서 함께 사라진다(좌표는 서버 navmesh 내부에만 존재).
  • 시계 신뢰: ServerNowUtcMs는 서버 권위 값이어야 한다. 서버 시계가 흔들리면 sequence/노후 판정이 오탐을 만든다.
  • 샘플링 기반 한계: 검사하지 않은 구간까지 수학적으로 보증하지 않는다. 보상/경쟁 등 고위험 행동에는 전 구간 검사 또는 표본 수 상향이 필요하다.
  • 선택지 B 의 기존 핸들러 결합. 같은 포트 혼합은 EchoServerHandler 의 소비/해제 규칙을 함께 고쳐야 완성된다(8.3). 이번 범위 밖이므로 별도 포트를 기본값으로 한다.
  • legacy 고정 길이 테이블은 관찰값이다. 클라이언트가 더 긴 패킷을 보낸다면 8.3 선택지 B 는 프레임을 잘라 버린다. 선택지 A 채택 시 영향 없음.
  • 본문 미검증: 빌드·테스트를 수행하지 않았다. 제안 코드는 문서 수준이며 실제 적용 전 컴파일/계약 픽스처 검증이 선행되어야 한다. 4.5의 16진수 예시도 사람이 만든 값이므로 권위 있는 원천이 아니다.
